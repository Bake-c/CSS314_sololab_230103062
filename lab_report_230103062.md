```python
from dataclasses import dataclass
import multiprocessing as mp
import time

@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float
    max_ms: float

def fnv1a_hash(s):
    h = 14695981039346656037
    for b in s.encode('utf-8'):
        h = ((h ^ b) * 1099511628211) & 0xffffffffffffffff
    return h

def run_benchmark(fn, warmup_runs=3, timed_runs=9):
    for _ in range(warmup_runs):
        fn()
    samples = []
    for _ in range(timed_runs):
        t = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns() - t) / 1e6)
    samples.sort()
    n = len(samples)
    median = samples[n // 2] if n % 2 else (samples[n // 2 - 1] + samples[n // 2]) / 2
    return BenchStats(samples[0], median, samples[3*n//4] - samples[n//4], samples[-1])

def emit_verify_tag(task_id, student_seed, primary_metric, secondary_metric):
    hw = mp.cpu_count()
    payload = f'{task_id}|{student_seed}|{hw}|{primary_metric:.4f}|{secondary_metric:.4f}'
    return f'VERIFY_HASH[{task_id}]: {fnv1a_hash(payload):016X} (P_log={hw})'
```

```python
﻿import ctypes
import os
import struct


def get_topology():
    if os.name != 'nt':
        raise RuntimeError('Automatic topology detection currently supports Windows only')
    kernel = ctypes.WinDLL('kernel32', use_last_error=True)
    fn = kernel.GetLogicalProcessorInformationEx
    fn.argtypes = [ctypes.c_int, ctypes.c_void_p, ctypes.POINTER(ctypes.c_ulong)]
    fn.restype = ctypes.c_int
    size = ctypes.c_ulong(0)
    fn(0xffff, None, ctypes.byref(size))
    buf = ctypes.create_string_buffer(size.value)
    if not fn(0xffff, buf, ctypes.byref(size)):
        raise ctypes.WinError(ctypes.get_last_error())
    data = buf.raw[:size.value]
    pos = 0
    cores = 0
    caches = []
    while pos < len(data):
        relation, length = struct.unpack_from('<II', data, pos)
        if length < 8 or pos + length > len(data):
            raise RuntimeError('Invalid topology record')
        if relation == 0:
            cores += 1
        elif relation == 2:
            level, assoc, line, capacity, kind = struct.unpack_from('<BBHII', data, pos + 8)
            caches.append((level, kind, line, capacity))
        pos += length
    return cores, sorted(set(caches))

if __name__ == '__main__':
    print(get_topology())
```

```python
import argparse
import ctypes
import multiprocessing as mp
from multiprocessing import shared_memory
from harness import run_benchmark, emit_verify_tag
from topology import get_topology

MASK = 0xffffffffffffffff
MULT = 6364136223846793005

def fs_worker(name, slot, iters):
    shm = shared_memory.SharedMemory(name=name)
    arr = (ctypes.c_uint64 * 512).from_buffer(shm.buf)
    try:
        acc = arr[slot]
        for i in range(iters):
            acc = (acc * MULT + (i | 1)) & MASK
            arr[slot] = acc
    finally:
        del arr
        shm.close()

def trial(p, stride, total):
    shm = shared_memory.SharedMemory(create=True, size=4096)
    try:
        shm.buf[:] = b'\0' * 4096
        processes = [mp.Process(target=fs_worker, args=(shm.name, t * stride,
                      total // p + (t < total % p))) for t in range(p)]
        for process in processes:
            process.start()
        for process in processes:
            process.join()
        if any(process.exitcode != 0 for process in processes):
            raise RuntimeError('A worker process failed')
    finally:
        shm.close()
        shm.unlink()

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--iterations', type=int, default=16_000_000)
    parser.add_argument('--warmups', type=int, default=2)
    parser.add_argument('--runs', type=int, default=7)
    parser.add_argument('--seed', type=int, default=230103062, help='Your numeric student ID')
    args = parser.parse_args()
    phys, _ = get_topology()
    logical = mp.cpu_count()
    print(f'physical={phys}, logical={logical}, iterations={args.iterations}, warmups={args.warmups}, runs={args.runs}', flush=True)
    measurements = {}
    def measure(p, stride):
        key = (p, stride)
        if key not in measurements:
            stats = run_benchmark(lambda: trial(p, stride, args.iterations), args.warmups, args.runs)
            measurements[key] = stats
            print(f'MEASURE p={p} stride={stride} median_ms={stats.median_ms:.6f} iqr_ms={stats.iqr_ms:.6f} throughput_Mops_s={args.iterations / stats.median_ms / 1000:.6f}', flush=True)
        return measurements[key]
    for stride in (1, 2, 4, 8, 16, 32):
        measure(phys, stride)
    counts = []
    p = 1
    while p <= logical:
        counts.append(p)
        p *= 2
    if counts[-1] != logical:
        counts.append(logical)
    for stride in (1, 16):
        t1 = measure(1, stride).median_ms
        for p in counts:
            tp = measure(p, stride).median_ms
            speedup = t1 / tp
            e = (1 / speedup - 1 / p) / (1 - 1 / p) if p > 1 else float('nan')
            print(f'SCALING stride={stride} p={p} median_ms={tp:.6f} speedup={speedup:.6f} efficiency={speedup/p:.6f} karp_flatt={e:.6f}', flush=True)
    print(emit_verify_tag('PHASE1', args.seed, measure(phys, 1).median_ms, measure(phys, 16).median_ms), flush=True)

if __name__ == '__main__':
    mp.freeze_support()
    main()
```

```python
import argparse
import math
import multiprocessing as mp
from harness import run_benchmark, emit_verify_tag

def serial_stage(n):
    result = 0.0
    for i in range(n // 20):
        x = (i % 997 + 1) / 997.0
        for _ in range(5):
            x = math.sqrt(abs(math.sin(x)) + 0.1)
        result += x
    return result

def compute_range(bounds):
    start, end = bounds
    result = 0.0
    for i in range(start, end):
        x = (i % 997 + 1) / 997.0
        # One multiply, one add, and one sin per iteration: 30 operations total.
        for _ in range(10):
            x = math.sin(x * 0.999 + 0.001)
        result += x
    return result

def pipeline(pool, p, n):
    a = serial_stage(n)
    bounds = [(i*n//p, (i+1)*n//p) for i in range(p)]
    return a + sum(pool.map(compute_range, bounds))

def main():
```

```python
def copy_worker(job):
```

```python
import argparse
import ctypes
import multiprocessing as mp
from multiprocessing import shared_memory
from harness import run_benchmark, emit_verify_tag
from topology import get_topology

def gemm_worker(names, n, start, end, variant, bs):
    shms = [shared_memory.SharedMemory(name=x) for x in names]
    a, b, c = [(ctypes.c_double * (n*n)).from_buffer(s.buf) for s in shms]
    try:
        if variant == 'naive':
            for i in range(start,end):
                for j in range(n):
                    total = 0.0
                    for k in range(n):
                        total += a[i*n+k] * b[k*n+j]
                    c[i*n+j] = total
        elif variant == 'reordered':
            for i in range(start,end):
                for k in range(n):
                    aik = a[i*n+k]
                    for j in range(n):
                        c[i*n+j] += aik * b[k*n+j]
        else:
            for ii in range(start,end,bs):
                for jj in range(0,n,bs):
                    for kk in range(0,n,bs):
                        for i in range(ii,min(ii+bs,end)):
                            for k in range(kk,min(kk+bs,n)):
                                aik = a[i*n+k]
                                for j in range(jj,min(jj+bs,n)):
                                    c[i*n+j] += aik * b[k*n+j]
    finally:
        del a,b,c
        for s in shms: s.close()

def execute(names, cshm, n, p, variant, bs):
    cshm.buf[:] = bytes(n*n*8)
    ps = [mp.Process(target=gemm_worker, args=(names,n,t*n//p,(t+1)*n//p,variant,bs)) for t in range(p)]
    for proc in ps: proc.start()
    for proc in ps: proc.join()
    if any(proc.exitcode != 0 for proc in ps): raise RuntimeError('GEMM worker failed')

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--n', type=int, default=256)
    ap.add_argument('--runs', type=int, default=7)
    ap.add_argument('--warmups', type=int, default=2)
    ap.add_argument('--seed', type=int, default=230103062, help='Your student ID')
    args = ap.parse_args()
    n = args.n
    p, _ = get_topology()
    shms = [shared_memory.SharedMemory(create=True,size=n*n*8) for _ in range(3)]
    try:
        a, b = [(ctypes.c_double*(n*n)).from_buffer(s.buf) for s in shms[:2]]
        for i in range(n*n):
            a[i] = ((i*7)%101)/101.0
            b[i] = ((i*13)%103)/103.0
        del a,b
        names = [s.name for s in shms]
        results = {}
        reference = None
        for variant, bs in [('naive',0), ('reordered',0)] + [('blocked',x) for x in (8,16,32,64,96,128)]:
            stats = run_benchmark(lambda: execute(names,shms[2],n,p,variant,bs), args.warmups,args.runs)
            out = (ctypes.c_double*(n*n)).from_buffer(shms[2].buf)
            if reference is None:
                reference = list(out)
            error = max((abs(out[i]-reference[i]) for i in range(n*n)),default=0)
            if variant == 'reordered':
                # Check naive against reordered, then compare each tiled variant
                # directly against reordered as required by the lab.
                reference = list(out)
            del out
            if error is not None and error >= 1e-9: raise RuntimeError(f'GEMM mismatch: {error}')
            gflops = 2*n**3/(stats.median_ms*1e6)
            results[variant,bs] = gflops
            print(f'{variant} bs={bs} ms={stats.median_ms:.6f} iqr_ms={stats.iqr_ms:.6f} gflops={gflops:.9f} tile_bytes={3*bs*bs*8} max_error={error}', flush=True)
        best = max((8,16,32,64,96,128),key=lambda x: results['blocked',x])
        print(f'Best tile: {best}',flush=True)
        print(emit_verify_tag('PHASE3',args.seed,results['naive',0],results['blocked',best]),flush=True)
    finally:
        for s in shms:
            s.close()
            s.unlink()

if __name__ == '__main__':
    mp.freeze_support()
    main()
```

### task4_scheduling.py

```python
import argparse
import math
import multiprocessing as mp
import statistics
import time
from harness import emit_verify_tag
from topology import get_topology

def item(i):
    x = 0.1
    for _ in range(1 + (i * i) // 1600):
        x = math.sqrt(abs(math.sin(x) * math.cos(x)) + 0.1)
    return x

def worker(t, p, n, strategy, chunk, counter, lock, times, sums):
    start = time.perf_counter_ns()
    result = 0.0
    if strategy == 'block':
        for i in range(t*n//p, (t+1)*n//p):
            result += item(i)
    elif strategy == 'cyclic':
        for i in range(t, n, p):
            result += item(i)
    else:
        while True:
            with lock:
                first = counter.value
                counter.value += chunk
            if first >= n:
                break
            for i in range(first, min(first+chunk, n)):
                result += item(i)
    times[t] = (time.perf_counter_ns() - start) / 1e6
    sums[t] = result

def trial(p, n, strategy, chunk):
    counter = mp.RawValue('i', 0)
    lock = mp.Lock()
    times = mp.Array('d', p, lock=False)
    sums = mp.Array('d', p, lock=False)
    ps = [mp.Process(target=worker, args=(t, p, n, strategy, chunk, counter, lock, times, sums)) for t in range(p)]
    start = time.perf_counter_ns()
    for proc in ps:
        proc.start()
    for proc in ps:
        proc.join()
    wall = (time.perf_counter_ns() - start) / 1e6
    if any(proc.exitcode != 0 for proc in ps):
        raise RuntimeError('Worker failed')
    values = list(times)
    return wall, max(values) / statistics.mean(values), sum(sums)

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--items', type=int, default=8000)
    ap.add_argument('--runs', type=int, default=1)
    ap.add_argument('--seed', type=int, default=230103062, help='Your student ID')
    args = ap.parse_args()
    p, _ = get_topology()
    results = {}
    for strategy, chunk in [('block', 0), ('cyclic', 0)] + [('dynamic', c) for c in (1,8,64,512)]:
        samples = [trial(p, args.items, strategy, chunk) for _ in range(args.runs)]
        # Select the median observed run so its wall time and imbalance describe the same run.
        samples.sort(key=lambda row: row[0])
        wall, imbalance, checksum = samples[len(samples)//2]
        results[strategy, chunk] = wall
        print(f'{strategy} chunk={chunk} wall_ms={wall:.6f} imbalance={imbalance:.6f} result_sum={checksum:.9f}', flush=True)
    best = min((c for c in (1,8,64,512)), key=lambda c: results['dynamic', c])
    print(f'Best dynamic chunk: {best}', flush=True)
    print(emit_verify_tag('PHASE4', args.seed, results['block',0], results['dynamic',best]), flush=True)

if __name__ == '__main__':
    mp.freeze_support()
    main()
```
